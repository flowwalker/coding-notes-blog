手写数据结构·线性表06：顺序串

顺序串（SeqString）

template<size_t N>
class SeqString {
    static_assert(N < numeric_limits<size_t>::max(), "invalid capacity");
    char data_[N + 1]{};
    size_t length_ = 0;

public:
    SeqString() = default;

    explicit SeqString(const char* s) {
        while (s[length_] != '\0') {
            if (length_ == N) throw length_error("SeqString capacity exceeded");
            data_[length_] = s[length_];
            ++length_;
        }
        data_[length_] = '\0';
    }

    bool empty() const { return length_ == 0; }
    size_t size() const { return length_; }
    size_t length() const { return length_; }
    constexpr size_t capacity() const { return N; }
    const char* c_str() const { return data_; }

    char& operator[](size_t i) { return data_[i]; }
    const char& operator[](size_t i) const { return data_[i]; }

    bool append(const SeqString& other) {
        if (other.length_ > N - length_) return false;
        for (size_t i = 0; i < other.length_; ++i)
            data_[length_ + i] = other.data_[i];
        length_ += other.length_;
        data_[length_] = '\0';
        return true;
    }

    bool insert(size_t pos, const SeqString& other) {
        if (pos > length_ || other.length_ > N - length_) return false;
        if (this == &other) {
            SeqString snapshot(other);
            return insert(pos, snapshot);
        }
        for (size_t i = length_; i > pos; --i)
            data_[i + other.length_ - 1] = data_[i - 1];
        for (size_t i = 0; i < other.length_; ++i)
            data_[pos + i] = other.data_[i];
        length_ += other.length_;
        data_[length_] = '\0';
        return true;
    }

    bool erase(size_t pos, size_t count) {
        if (pos > length_) return false;
        if (count > length_ - pos) count = length_ - pos;
        for (size_t i = pos; i + count < length_; ++i)
            data_[i] = data_[i + count];
        length_ -= count;
        data_[length_] = '\0';
        return true;
    }

    SeqString substr(size_t pos, size_t count) const {
        if (pos > length_) throw out_of_range("SeqString::substr");
        if (count > length_ - pos) count = length_ - pos;
        SeqString out;
        for (size_t i = 0; i < count; ++i) out.data_[i] = data_[pos + i];
        out.length_ = count;
        out.data_[count] = '\0';
        return out;
    }

    ptrdiff_t find(const SeqString& pat) const {
        if (pat.length_ == 0) return 0;
        if (pat.length_ > length_) return -1;
        for (size_t i = 0; i + pat.length_ <= length_; ++i) {
            size_t j = 0;
            while (j < pat.length_ && data_[i + j] == pat.data_[j]) ++j;
            if (j == pat.length_) return static_cast<ptrdiff_t>(i);
        }
        return -1;
    }

    bool operator==(const SeqString& other) const {
        if (length_ != other.length_) return false;
        for (size_t i = 0; i < length_; ++i)
            if (data_[i] != other.data_[i]) return false;
        return true;
    }

    void clear() {
        length_ = 0;
        data_[0] = '\0';
    }
};

工程细节

  • data_ 有 N + 1 个位置，额外位置留给 \0；length_ 不包含终止字符。修改长度后重新写入 data_[length_] = '\0'。
  • 插入先从后往前移动原字符。插入自身时先复制快照，再按快照插入，避免原串被移动后污染来源。
  • 容量检查写成“待插入长度是否超过剩余容量”，避免长度相加溢出。删除和截取会把过大的 count 截断到剩余长度。
  • find() 为朴素匹配，空模式返回 0，未找到返回 -1。下标访问不检查边界，不能改写终止字符；构造参数必须指向有效、以 \0 结尾的字符序列。
